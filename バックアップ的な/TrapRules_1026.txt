extends RefCounted


# 移動した駒が踏んだ、または通過した罠を取得する
# 自分が配置した罠も発動候補に含める
static func get_triggered_traps(
	pieces: Dictionary,
	from_square: Vector2i,
	to_square: Vector2i,
	moving_piece: Dictionary
) -> Array[Vector2i]:
	var triggered_traps: Array[Vector2i] = []

	var dx: int = to_square.x - from_square.x
	var dy: int = to_square.y - from_square.y

	# 直線または斜めの移動なら、通過するマスも調べる
	var is_straight_path: bool = (
		dx == 0
		or dy == 0
		or absi(dx) == absi(dy)
	)

	if is_straight_path:
		var step_x: int = 0
		var step_y: int = 0

		if dx > 0:
			step_x = 1
		elif dx < 0:
			step_x = -1

		if dy > 0:
			step_y = 1
		elif dy < 0:
			step_y = -1

		var current_square: Vector2i = (
			from_square + Vector2i(step_x, step_y)
		)

		while true:
			if pieces.has(current_square):
				var piece: Dictionary = pieces[current_square]

				# 所有者に関係なく、罠を発動候補にする
				if str(piece.get("category", "")) == "罠":
					triggered_traps.append(current_square)

			if current_square == to_square:
				break

			current_square += Vector2i(step_x, step_y)

	else:
		# 跳躍など、通過マスを定義しにくい移動は到着マスだけ調べる
		if pieces.has(to_square):
			var piece: Dictionary = pieces[to_square]

			if str(piece.get("category", "")) == "罠":
				triggered_traps.append(to_square)

	return triggered_traps
# 罠の効果を判定する
# 実際の盤面変更は呼び出し側に任せる
static func resolve_trap_effect(
	trap_piece: Dictionary,
	moving_piece: Dictionary
) -> Dictionary:
	# 罠以外は対象外
	if str(trap_piece.get("category", "")) != "罠":
		return {"triggered": false}

	# 兵と王を罠の対象にする
	var moving_category: String = str(
		moving_piece.get("category", "")
	)

	if moving_category != "兵" and moving_category != "王" and moving_category !="成":
		return {"triggered": false}

	# 罠の効果を判定
	match str(trap_piece.get("trap_effect", "")):
		"destroy_soldier":
			return {
				"triggered": true,
				"consume_trap": true,
				"piece_outcome": "destroy",
				"message": "落とし穴が発動しました。"
			}

		_:
			return {"triggered": false}
